  2. Create retry wrapper with exponential backoff:
     ```javascript
     async _withRetry(fn, context = {}) {
       let lastError;
       for (let attempt = 0; attempt <= this.maxRetries; attempt++) {
         try {
           return await fn();
         } catch (error) {
           lastError = error;

           if (attempt < this.maxRetries) {
             const delay = Math.min(
               this.baseDelay * Math.pow(2, attempt),
               this.maxDelay
             );

             this.emit('retry', {
               ...context,
               attempt: attempt + 1,
               maxRetries: this.maxRetries,
               delay,
               error: error.message
             });

             await new Promise(r => setTimeout(r, delay));
           }
         }
       }
       throw lastError;
     }
     ```

  3. Update _processDocument to use retry:
     ```javascript
     async _processDocument(doc) {
       const docId = doc.id;
       this.emit('document:start', { docId, title: doc.title });

       try {
         await this._withRetry(
           async () => {
             const pdfBuffer = await this._downloadPdf(docId);
             const images = await this.pdfRenderer.renderBuffer(pdfBuffer, {
               dpi: this.dpi, docId
             });
             const base64Images = images.map(img => img.base64);

             return await this.ingestionManager.ingestDocument(docId, doc.original_file_name, {
               base64Images,
               metadata: { title: doc.title, tags: doc.tags }
             });
           },
           { docId }
         );

         this.stats.processed++;
         this.stats.succeeded++;
         this.emit('document:success', { docId });
         this.emit('progress', this._getProgress());

       } catch (error) {
         // All retries exhausted
         this.stats.processed++;
         this.stats.failed++;
         this.errors.push({
           docId,
           error: error.message,
           retriesExhausted: true,
           timestamp: new Date()
         });
         this.emit('document:error', { docId, error: error.message, final: true });
         this.emit('progress', this._getProgress());
       }
     }
     ```

  4. Add retry stats to job result (totalRetries, avgRetriesPerDoc)

<task type="auto">
  <name>Task 2: Create batch ingestion API endpoints</name>
  <files>routes/api/visual-rag.js</files>
  <action>
  Add API endpoints for batch job management:

  ```javascript
  const { BatchIngestionJob } = require('../../services/visual-rag');

  // Store active jobs (in production, use Redis or database)
  const activeJobs = new Map();

  // POST /api/visual-rag/batch/start
  router.post('/batch/start', async (req, res) => {
    try {
      const {
        filters = {},
        options = {}
      } = req.body;

      const job = new BatchIngestionJob({
        concurrency: options.concurrency || 2,
        skipIngested: options.skipIngested ?? true,
        maxRetries: options.maxRetries || 3,
        dpi: options.dpi || 300
      });

      // Store job for monitoring
      activeJobs.set(job.jobId, job);

      // Start job in background (don't await)
      job.start(filters).finally(() => {
        // Keep completed jobs for 1 hour
        setTimeout(() => activeJobs.delete(job.jobId), 3600000);
      });

      res.json({
        success: true,
        jobId: job.jobId,
        message: 'Batch ingestion started'
      });
    } catch (error) {
      res.status(500).json({ success: false, error: error.message });
    }
  });

  // GET /api/visual-rag/batch/:jobId/status
  router.get('/batch/:jobId/status', (req, res) => {
    const job = activeJobs.get(req.params.jobId);
    if (!job) {
      return res.status(404).json({ error: 'Job not found' });
    }
    res.json(job._getProgress());
  });

  // POST /api/visual-rag/batch/:jobId/pause
  router.post('/batch/:jobId/pause', (req, res) => {
    const job = activeJobs.get(req.params.jobId);
    if (!job) {
      return res.status(404).json({ error: 'Job not found' });
    }
    job.pause();
    res.json({ success: true, status: job.status });
  });

  // POST /api/visual-rag/batch/:jobId/resume
  router.post('/batch/:jobId/resume', (req, res) => {
    const job = activeJobs.get(req.params.jobId);
    if (!job) {
      return res.status(404).json({ error: 'Job not found' });
    }
    job.resume();
    res.json({ success: true, status: job.status });
  });

  // POST /api/visual-rag/batch/:jobId/cancel
  router.post('/batch/:jobId/cancel', (req, res) => {
    const job = activeJobs.get(req.params.jobId);
    if (!job) {
      return res.status(404).json({ error: 'Job not found' });
    }
    job.cancel();
    res.json({ success: true, status: job.status });
  });

  // GET /api/visual-rag/batch/list
  router.get('/batch/list', (req, res) => {
    const jobs = Array.from(activeJobs.values()).map(job => ({
      jobId: job.jobId,
      status: job.status,
      stats: job.stats
    }));
    res.json({ jobs });
  });
  ```

  Add request validation and rate limiting for production.
