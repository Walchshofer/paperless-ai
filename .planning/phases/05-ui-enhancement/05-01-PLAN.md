  ```javascript
  /**
   * Visual RAG Overlay Configuration
   * Color codes and field mappings for domain-specific detection
   */

  const DOMAIN_COLORS = {
    FINANCIAL: {
      primary: '#F97316',  // Orange-500
      range: ['#FFF7ED', '#FDBA74', '#F97316', '#C2410C', '#9A3412']
    },
    MEDICAL: {
      primary: '#22C55E',  // Green-500
      range: ['#BBF7D0', '#86EFAC', '#22C55E', '#15803D', '#065F46']
    },
    LEGAL: {
      primary: '#A855F7',  // Purple-500
      range: ['#E9D5FF', '#C084FC', '#A855F7', '#7E22CE', '#581C87']
    },
    GENERAL: {
      primary: '#3B82F6',  // Blue-500
      range: ['#93C5FD', '#60A5FA', '#3B82F6', '#2563EB', '#1D4ED8']
    }
  };

  // Re-export DOMAIN_FIELD_SPECS
  const DOMAIN_FIELD_SPECS = { /* from Task 2 */ };

  function getColorForLabel(label, domain) {
    const spec = DOMAIN_FIELD_SPECS[domain?.toLowerCase()];
    if (!spec) return '#6B7280'; // Gray fallback

    const normalizedLabel = label.toLowerCase().replace(/[\s\/]/g, '_');
    const field = Object.values(spec.fields).find(
      f => f.label.toLowerCase().replace(/[\s\/]/g, '_') === normalizedLabel
    );
    return field?.color || DOMAIN_COLORS[spec.name]?.primary || '#6B7280';
  }

  function getPaperlessMapping(label, domain) {
    const spec = DOMAIN_FIELD_SPECS[domain?.toLowerCase()];
    if (!spec) return null;

    const normalizedLabel = label.toLowerCase().replace(/[\s\/]/g, '_');
    const field = Object.values(spec.fields).find(
      f => f.label.toLowerCase().replace(/[\s\/]/g, '_') === normalizedLabel
    );
    return field?.mapping || null;
  }

  function isMandatoryField(label, domain) {
    const spec = DOMAIN_FIELD_SPECS[domain?.toLowerCase()];
    if (!spec) return false;

    const normalizedLabel = label.toLowerCase().replace(/[\s\/]/g, '_');
    return spec.mandatory.some(m => {
      const mandatoryField = spec.fields[m];
      return mandatoryField?.label.toLowerCase().replace(/[\s\/]/g, '_') === normalizedLabel;
    });
  }

  function getLegendForDomain(domain) {
    const spec = DOMAIN_FIELD_SPECS[domain?.toLowerCase()];
    if (!spec) return [];

    return Object.entries(spec.fields).map(([key, field]) => ({
      key,
      label: field.label,
      color: field.color,
      mapping: field.mapping,
      isMandatory: spec.mandatory.includes(key)
    }));
  }

  module.exports = {
    DOMAIN_COLORS,
    DOMAIN_FIELD_SPECS,
    getColorForLabel,
    getPaperlessMapping,
    isMandatoryField,
    getLegendForDomain
  };
  ```

  Update OverlayExtractor to import and use these helpers.
