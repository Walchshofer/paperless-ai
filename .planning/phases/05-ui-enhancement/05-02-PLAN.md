  ```javascript
  /**
   * OverlayViewer - Renders visual overlays on document images
   * Supports color-coded bounding boxes with domain-specific styling
   */
  class OverlayViewer {
    constructor(containerEl, options = {}) {
      this.container = typeof containerEl === 'string'
        ? document.querySelector(containerEl)
        : containerEl;
      this.imageEl = null;
      this.overlayCanvas = null;
      this.overlays = [];
      this.domain = options.domain || 'GENERAL';
      this.showMandatoryOnly = options.showMandatoryOnly || false;
      this.onOverlayClick = options.onOverlayClick || null;
      this.scale = 1;

      this._init();
    }

    _init() {
      // Create wrapper with relative positioning
      this.wrapper = document.createElement('div');
      this.wrapper.className = 'overlay-viewer-wrapper relative';
      this.container.appendChild(this.wrapper);

      // Create canvas for overlays
      this.overlayCanvas = document.createElement('canvas');
      this.overlayCanvas.className = 'overlay-canvas absolute top-0 left-0 pointer-events-auto';
      this.wrapper.appendChild(this.overlayCanvas);

      // Create tooltip element
      this.tooltip = document.createElement('div');
      this.tooltip.className = 'overlay-tooltip hidden absolute bg-gray-900 text-white px-3 py-2 rounded-lg text-sm shadow-lg z-50';
      this.wrapper.appendChild(this.tooltip);

      // Add event listeners
      this.overlayCanvas.addEventListener('mousemove', this._handleMouseMove.bind(this));
      this.overlayCanvas.addEventListener('mouseleave', this._hideTooltip.bind(this));
      this.overlayCanvas.addEventListener('click', this._handleClick.bind(this));
    }

    setImage(imgEl) {
      this.imageEl = imgEl;
      this.wrapper.insertBefore(imgEl, this.overlayCanvas);
      this._resizeCanvas();
      this._render();
    }

    setOverlays(overlays, domain = null) {
      this.overlays = overlays || [];
      if (domain) this.domain = domain;
      this._render();
    }

    _resizeCanvas() {
      if (!this.imageEl) return;
      const rect = this.imageEl.getBoundingClientRect();
      this.overlayCanvas.width = rect.width;
      this.overlayCanvas.height = rect.height;
      this.scale = rect.width / 1000; // Coordinates are 0-1000
    }

    _render() {
      const ctx = this.overlayCanvas.getContext('2d');
      ctx.clearRect(0, 0, this.overlayCanvas.width, this.overlayCanvas.height);

      const filtered = this.showMandatoryOnly
        ? this.overlays.filter(o => o.isMandatory)
        : this.overlays;

      filtered.forEach(overlay => {
        this._drawOverlay(ctx, overlay);
      });
    }

    _drawOverlay(ctx, overlay) {
      const { boundingBox, color, isMandatory, label } = overlay;
      const x = boundingBox.x * this.scale;
      const y = boundingBox.y * this.scale;
      const w = boundingBox.width * this.scale;
      const h = boundingBox.height * this.scale;

      // Draw box
      ctx.strokeStyle = color;
      ctx.lineWidth = isMandatory ? 3 : 2;
      ctx.setLineDash(isMandatory ? [] : [5, 3]);
      ctx.strokeRect(x, y, w, h);

      // Draw label background
      ctx.fillStyle = color;
      const labelWidth = ctx.measureText(label).width + 8;
      ctx.fillRect(x, y - 20, labelWidth, 18);

      // Draw label text
      ctx.fillStyle = '#FFFFFF';
      ctx.font = '12px Inter, sans-serif';
      ctx.fillText(label, x + 4, y - 6);

      // Mandatory indicator
      if (isMandatory) {
        ctx.fillStyle = color;
        ctx.beginPath();
        ctx.arc(x + w - 8, y + 8, 6, 0, Math.PI * 2);
        ctx.fill();
        ctx.fillStyle = '#FFFFFF';
        ctx.font = 'bold 10px sans-serif';
        ctx.fillText('!', x + w - 11, y + 12);
      }

      ctx.setLineDash([]);
    }

    _handleMouseMove(e) {
      const rect = this.overlayCanvas.getBoundingClientRect();
      const x = (e.clientX - rect.left) / this.scale;
      const y = (e.clientY - rect.top) / this.scale;

      const hovered = this.overlays.find(o => this._isPointInBox(x, y, o.boundingBox));

      if (hovered) {
        this._showTooltip(e, hovered);
      } else {
        this._hideTooltip();
      }
    }

    _isPointInBox(x, y, box) {
      return x >= box.x && x <= box.x + box.width &&
             y >= box.y && y <= box.y + box.height;
    }

    _showTooltip(e, overlay) {
      const { label, value, paperlessMapping, confidence, isMandatory } = overlay;
      this.tooltip.innerHTML = `
        <div class="font-semibold">${label}${isMandatory ? ' ⚠️' : ''}</div>
        ${value ? `<div class="text-gray-300">"${value}"</div>` : ''}
        <div class="text-xs text-gray-400 mt-1">
          → ${paperlessMapping || 'unmapped'}
          <span class="ml-2">${Math.round(confidence * 100)}%</span>
        </div>
      `;
      this.tooltip.style.left = `${e.offsetX + 10}px`;
      this.tooltip.style.top = `${e.offsetY + 10}px`;
      this.tooltip.classList.remove('hidden');
    }

    _hideTooltip() {
      this.tooltip.classList.add('hidden');
    }

    _handleClick(e) {
      if (!this.onOverlayClick) return;
      const rect = this.overlayCanvas.getBoundingClientRect();
      const x = (e.clientX - rect.left) / this.scale;
      const y = (e.clientY - rect.top) / this.scale;

      const clicked = this.overlays.find(o => this._isPointInBox(x, y, o.boundingBox));
      if (clicked) {
        this.onOverlayClick(clicked);
      }
    }

    toggleMandatoryOnly(show) {
      this.showMandatoryOnly = show;
      this._render();
    }

    destroy() {
      this.wrapper.remove();
    }
  }

  // Export for module systems
  if (typeof module !== 'undefined') module.exports = { OverlayViewer };
  ```

  ```javascript
  /**
   * OverlayLegend - Displays domain-specific field legend
   */
  class OverlayLegend {
    constructor(containerEl, options = {}) {
      this.container = typeof containerEl === 'string'
        ? document.querySelector(containerEl)
        : containerEl;
      this.domain = options.domain || 'GENERAL';
      this.onFilterChange = options.onFilterChange || null;
      this.activeFilters = new Set();

      this._render();
    }

    setDomain(domain) {
      this.domain = domain;
      this.activeFilters.clear();
      this._render();
    }

    async _render() {
      // Fetch legend data from API
      const response = await fetch(`/api/visual-rag/legend/${this.domain.toLowerCase()}`);
      const legend = await response.json();

      this.container.innerHTML = `
        <div class="overlay-legend p-4 bg-gray-50 dark:bg-gray-800 rounded-lg">
          <div class="flex items-center justify-between mb-3">
            <h4 class="font-semibold text-sm text-gray-700 dark:text-gray-300">
              ${this._getDomainIcon()} ${this.domain} Fields
            </h4>
            <button class="text-xs text-gray-500 hover:text-gray-700" id="legend-toggle-all">
              Toggle All
            </button>
          </div>
          <div class="grid grid-cols-2 gap-2">
            ${legend.map(field => this._renderField(field)).join('')}
          </div>
          <div class="mt-3 pt-3 border-t border-gray-200 dark:border-gray-700">
            <label class="flex items-center text-xs text-gray-600 dark:text-gray-400">
              <input type="checkbox" id="show-mandatory-only" class="mr-2 rounded">
              Show mandatory only
            </label>
          </div>
        </div>
      `;

      this._attachListeners();
    }

    _renderField(field) {
      const { key, label, color, isMandatory } = field;
      return `
        <label class="flex items-center text-xs cursor-pointer hover:bg-gray-100 dark:hover:bg-gray-700 p-1 rounded"
               data-field="${key}">
          <span class="w-3 h-3 rounded mr-2 flex-shrink-0"
                style="background-color: ${color}"></span>
          <span class="truncate ${isMandatory ? 'font-semibold' : ''}">${label}</span>
          ${isMandatory ? '<span class="ml-1 text-orange-500">*</span>' : ''}
        </label>
      `;
    }

    _getDomainIcon() {
      const icons = {
        FINANCIAL: '🟧',
        MEDICAL: '🟩',
        LEGAL: '🟪',
        GENERAL: '🟦'
      };
      return icons[this.domain] || '📄';
    }

    _attachListeners() {
      this.container.querySelector('#show-mandatory-only')?.addEventListener('change', (e) => {
        if (this.onFilterChange) {
          this.onFilterChange({ mandatoryOnly: e.target.checked });
        }
      });
    }
  }

  if (typeof module !== 'undefined') module.exports = { OverlayLegend };
  ```

  ```javascript
  const { getLegendForDomain, DOMAIN_FIELD_SPECS } = require('../../services/visual-rag/overlayConfig');
  const { visualOverlayRepository } = require('../../services/visual-rag');

  // GET /api/visual-rag/overlays/:docId
  router.get('/overlays/:docId', async (req, res) => {
    try {
      const docId = parseInt(req.params.docId, 10);
      const { page } = req.query;

      let overlays;
      if (page) {
        overlays = await visualOverlayRepository.getByDocIdAndPage(docId, parseInt(page, 10));
      } else {
        overlays = await visualOverlayRepository.getByDocId(docId);
      }

      // Transform to UI format
      const formatted = overlays.map(o => ({
        id: o.id,
        label: o.overlayData?.label || o.semanticLabel,
        value: o.overlayData?.text || null,
        domain: o.overlayData?.domain || 'GENERAL',
        color: o.overlayData?.color || '#6B7280',
        boundingBox: o.overlayData?.boundingBox || {
          x: o.overlayData?.box?.[1] || 0,
          y: o.overlayData?.box?.[0] || 0,
          width: (o.overlayData?.box?.[3] - o.overlayData?.box?.[1]) || 0,
          height: (o.overlayData?.box?.[2] - o.overlayData?.box?.[0]) || 0
        },
        paperlessMapping: o.overlayData?.paperlessMapping || null,
        isMandatory: o.overlayData?.isMandatory || false,
        confidence: o.overlayData?.confidence || o.confidence || 0.5,
        pageNumber: o.pageNumber
      }));

      res.json({
        docId,
        overlays: formatted,
        count: formatted.length
      });
    } catch (error) {
      res.status(500).json({ error: error.message });
    }
  });

  // GET /api/visual-rag/legend/:domain
  router.get('/legend/:domain', (req, res) => {
    const domain = req.params.domain.toLowerCase();
    const legend = getLegendForDomain(domain);

    if (!legend || legend.length === 0) {
      return res.status(404).json({ error: 'Domain not found' });
    }

    res.json(legend);
  });

  // GET /api/visual-rag/domains
  router.get('/domains', (req, res) => {
    const domains = Object.keys(DOMAIN_FIELD_SPECS).map(key => ({
      key,
      name: DOMAIN_FIELD_SPECS[key].name,
      fieldCount: Object.keys(DOMAIN_FIELD_SPECS[key].fields).length,
      mandatoryCount: DOMAIN_FIELD_SPECS[key].mandatory.length
    }));

    res.json({ domains });
  });
  ```
