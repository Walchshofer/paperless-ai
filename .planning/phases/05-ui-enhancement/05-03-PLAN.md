     ```javascript
     let overlayViewer = null;
     let overlayLegend = null;

     async function loadDocumentWithOverlays(docId) {
       // Load document preview
       const previewImg = document.getElementById('document-preview');
       previewImg.src = `/manual/preview/${docId}`;

       // Wait for image load
       previewImg.onload = async () => {
         // Initialize overlay viewer
         if (!overlayViewer) {
           overlayViewer = new OverlayViewer('#overlay-container', {
             onOverlayClick: (overlay) => {
               console.log('Clicked:', overlay.label, overlay.value);
               // Could populate form fields here
             }
           });
         }

         // Fetch overlays
         const response = await fetch(`/api/visual-rag/overlays/${docId}`);
         const data = await response.json();

         // Determine domain from overlays
         const domain = data.overlays[0]?.domain || 'GENERAL';

         // Set image and overlays
         overlayViewer.setImage(previewImg);
         overlayViewer.setOverlays(data.overlays, domain);

         // Initialize legend
         if (!overlayLegend) {
           overlayLegend = new OverlayLegend('#overlay-legend', {
             domain: domain,
             onFilterChange: ({ mandatoryOnly }) => {
               overlayViewer.toggleMandatoryOnly(mandatoryOnly);
             }
           });
         } else {
           overlayLegend.setDomain(domain);
         }
       };
     }
     ```

  2. **Fetch and display overlay badges after table draw:**
     ```javascript
     async function loadOverlayBadges() {
       const rows = document.querySelectorAll('[id^="overlay-badge-"]');

       for (const row of rows) {
         const docId = row.id.replace('overlay-badge-', '');
         try {
           const response = await fetch(`/api/visual-rag/overlays/${docId}`);
           const { overlays, count } = await response.json();

           if (count === 0) {
             row.innerHTML = '<span class="text-gray-400 text-xs">None</span>';
             continue;
           }

           // Group by domain
           const domains = [...new Set(overlays.map(o => o.domain))];
           const mandatory = overlays.filter(o => o.isMandatory).length;

           row.innerHTML = domains.map(d => `
             <span class="inline-flex items-center px-2 py-0.5 rounded text-xs font-medium"
                   style="background-color: ${getDomainColor(d)}20; color: ${getDomainColor(d)}">
               ${getDomainIcon(d)} ${overlays.filter(o => o.domain === d).length}
             </span>
           `).join(' ') + (mandatory > 0 ? `
             <span class="text-orange-500 text-xs ml-1" title="${mandatory} mandatory fields">
               ⚠️${mandatory}
             </span>
           ` : '');
         } catch (e) {
           row.innerHTML = '<span class="text-red-400 text-xs">Error</span>';
         }
       }
     }

     function getDomainColor(domain) {
       const colors = { FINANCIAL: '#F97316', MEDICAL: '#22C55E', LEGAL: '#A855F7', GENERAL: '#3B82F6' };
       return colors[domain] || '#6B7280';
     }

     function getDomainIcon(domain) {
       const icons = { FINANCIAL: '🟧', MEDICAL: '🟩', LEGAL: '🟪', GENERAL: '🟦' };
       return icons[domain] || '📄';
     }
     ```

  3. **Update reanalyze button to trigger re-ingestion:**
     ```javascript
     async function reanalyzeDocument(docId) {
       const btn = document.querySelector(`[data-reanalyze="${docId}"]`);
       btn.disabled = true;
       btn.innerHTML = '<span class="animate-spin">⟳</span> Processing...';

       try {
         // Trigger visual RAG re-ingestion
         const response = await fetch(`/api/visual-rag/ingest/${docId}`, {
           method: 'POST',
           headers: { 'Content-Type': 'application/json' },
           body: JSON.stringify({ force: true })
         });

         const result = await response.json();

         if (result.success) {
           btn.innerHTML = '✓ Done';
           // Refresh overlay badges
           loadOverlayBadges();
           showToast(`Detected ${result.overlayCount} fields`, 'success');
         } else {
           throw new Error(result.error);
         }
       } catch (e) {
         btn.innerHTML = '✗ Failed';
         showToast(`Reanalyze failed: ${e.message}`, 'error');
       } finally {
         setTimeout(() => {
           btn.disabled = false;
           btn.innerHTML = '🔄 Reanalyze';
         }, 3000);
       }
     }
     ```

  4. **Add ingest endpoint to API:**
     ```javascript
     // routes/api/visual-rag.js
     router.post('/ingest/:docId', async (req, res) => {
       const docId = parseInt(req.params.docId, 10);
       const { force } = req.body;

       try {
         // Fetch document from paperless
         const doc = await paperlessService.getDocument(docId);

         // Download and render PDF
         const pdfBuffer = await paperlessService.downloadDocument(docId);
         const images = await pdfRenderer.renderBuffer(pdfBuffer, { dpi: 300, docId });

         // Ingest with visual RAG
         const result = await ingestionManager.ingestDocument(docId, doc.original_file_name, {
           base64Images: images.map(i => i.base64),
           metadata: { title: doc.title, tags: doc.tags }
         });

         res.json({
           success: true,
           overlayCount: result.overlayExtraction?.overlayCount || 0,
           domain: result.overlayExtraction?.domain || 'general'
         });
       } catch (error) {
         res.status(500).json({ success: false, error: error.message });
       }
     });
     ```
