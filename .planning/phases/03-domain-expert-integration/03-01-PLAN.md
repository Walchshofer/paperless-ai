  Create a centralized DomainResolver service that determines document domain from multiple signals:

  ```javascript
  const DOMAIN_TYPES = {
    MEDICAL: 'medical',
    FINANCIAL: 'financial',
    LEGAL: 'legal',
    GENERAL: 'general'
  };

  class DomainResolver {
    constructor(options = {}) {
      this.paperlessService = options.paperlessService || paperlessService;
    }

    /**
     * Resolve domain from multiple signals
     * Priority: explicit > classification > document_type > tags > content
     */
    async resolveDomain(docId, options = {}) {
      const { explicit, classificationResult, documentType, tags, content } = options;

      // 1. Explicit override
      if (explicit && DOMAIN_TYPES[explicit.toUpperCase()]) {
        return explicit.toLowerCase();
      }

      // 2. From classification result (ExpertPipelineExecutor)
      if (classificationResult?.domain) {
        return this._normalizeDomain(classificationResult.domain);
      }

      // 3. From document type
      if (documentType) {
        const domain = this._domainFromDocType(documentType);
        if (domain !== 'general') return domain;
      }

      // 4. From tags
      if (tags?.length > 0) {
        const domain = this._domainFromTags(tags);
        if (domain !== 'general') return domain;
      }

      // 5. From content keywords
      if (content) {
        return this._domainFromContent(content);
      }

      return 'general';
    }

    _domainFromDocType(docType) {
      // Map document types to domains
      const typeMap = {
        'invoice': 'financial', 'receipt': 'financial', 'bill': 'financial',
        'lab_report': 'medical', 'prescription': 'medical', 'referral': 'medical',
        'contract': 'legal', 'agreement': 'legal', 'lease': 'legal'
      };
      return typeMap[docType?.toLowerCase()] || 'general';
    }

    _domainFromTags(tags) {
      // Check tag names for domain keywords
      const tagNames = tags.map(t => (t.name || t).toLowerCase());
      if (tagNames.some(t => /medical|health|lab|doctor|patient/.test(t))) return 'medical';
      if (tagNames.some(t => /invoice|financial|tax|vat|payment/.test(t))) return 'financial';
      if (tagNames.some(t => /legal|contract|agreement|court/.test(t))) return 'legal';
      return 'general';
    }

    _domainFromContent(content) {
      const text = content.toLowerCase();
      // Simple keyword scoring
      const scores = {
        medical: (text.match(/patient|diagnosis|medication|lab|blood|doctor|mg|ml/g) || []).length,
        financial: (text.match(/invoice|total|amount|tax|vat|payment|€|\$|iban/g) || []).length,
        legal: (text.match(/contract|agreement|party|clause|witness|notary|hereby/g) || []).length
      };
      const max = Math.max(...Object.values(scores));
      if (max < 3) return 'general';
      return Object.entries(scores).find(([k, v]) => v === max)?.[0] || 'general';
    }

    _normalizeDomain(domain) {
      const normalized = domain?.toLowerCase();
      return DOMAIN_TYPES[normalized?.toUpperCase()] ? normalized : 'general';
    }
  }
  ```
