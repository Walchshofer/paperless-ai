  Create OverlayRefiner that uses domain experts to validate and enrich overlay detections:

  ```javascript
  class OverlayRefiner {
    constructor(options = {}) {
      this.ollamaClient = options.ollamaClient || axios.create({
        baseURL: config.ollama?.apiUrl || 'http://localhost:11434',
        timeout: 30000
      });
    }

    /**
     * Refine overlays using domain expert validation
     * @param {Array} overlays - Raw overlays from OverlayExtractor
     * @param {string} domain - Document domain (medical, financial, legal)
     * @param {Object} context - Additional context (ocrText, documentType)
     * @returns {Promise<Array>} Refined overlays with adjusted confidence/labels
     */
    async refineOverlays(overlays, domain, context = {}) {
      if (!overlays?.length || domain === 'general') {
        return overlays; // No refinement for general domain
      }

      const model = this._getExpertModel(domain);
      const prompt = this._buildRefinementPrompt(overlays, domain, context);

      try {
        const response = await this._callExpert(model, prompt);
        return this._applyRefinements(overlays, response);
      } catch (error) {
        logger.warn(`[OverlayRefiner] Refinement failed: ${error.message}`);
        return overlays; // Return unrefined on failure
      }
    }

    _getExpertModel(domain) {
      // Map domain to appropriate expert model
      const models = {
        medical: config.expertModels?.medical?.analysis || 'medtext-llama3',
        financial: config.expertModels?.financial?.analysis || 'fino1-8b',
        legal: config.expertModels?.legal?.analysis || 'dragon-finance:latest'
      };
      return models[domain] || config.ollama?.model;
    }

    _buildRefinementPrompt(overlays, domain, context) {
      // Build domain-specific validation prompt
      return `You are a ${domain} document expert. Review these detected visual elements and:
1. Validate if each detection is correct for a ${domain} document
2. Suggest more specific labels (e.g., "signature" → "physician_signature")
3. Adjust confidence based on your domain knowledge

Detected elements:
${JSON.stringify(overlays, null, 2)}

Document text excerpt:
${context.ocrText?.substring(0, 500) || 'N/A'}

Output JSON array with refined elements:
[{"original_label": "...", "refined_label": "...", "confidence_adjustment": 0.1, "valid": true}]`;
    }

    async _callExpert(model, prompt) {
      const response = await this.ollamaClient.post('/api/generate', {
        model,
        prompt,
        stream: false,
        options: { temperature: 0.1, num_predict: 1024 }
      });
      return response.data.response;
    }

    _applyRefinements(overlays, response) {
      // Parse expert response and apply to overlays
      try {
        const refinements = JSON.parse(response.match(/\[[\s\S]*\]/)?.[0] || '[]');
        return overlays.map((overlay, i) => {
          const refinement = refinements[i];
          if (!refinement?.valid) {
            return { ...overlay, confidence: overlay.confidence * 0.5, refined: false };
          }
          return {
            ...overlay,
            label: refinement.refined_label || overlay.label,
            originalLabel: overlay.label,
            confidence: Math.min(1, overlay.confidence + (refinement.confidence_adjustment || 0)),
            refined: true,
            expertValidated: true
          };
        });
      } catch (e) {
        return overlays;
      }
    }
  }
  ```
