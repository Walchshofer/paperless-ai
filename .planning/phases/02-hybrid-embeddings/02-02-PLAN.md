  Integrate HybridSearchService into IngestionManager:

  1. Import HybridSearchService in IngestionManager
  2. Add hybridSearchService as constructor option with lazy initialization
  3. Add new method `hybridSearch(query, options)`:
     ```javascript
     async hybridSearch(query, options = {}) {
       const { k = 5, includeOverlays = true, mode = 'hybrid' } = options;

       // Mode: 'hybrid', 'visual', 'text'
       let results;
       switch (mode) {
         case 'visual':
           results = await this.visualSearch(query, { k });
           break;
         case 'text':
           results = await this.hybridSearchService.textSearch(query, k);
           break;
         default:
           results = await this.hybridSearchService.search(query, { k });
       }

       // Enrich with overlays if requested
       if (includeOverlays && results.results?.length > 0) {
         for (const result of results.results) {
           if (result.docId) {
             result.overlays = await this.getOverlaysForPage(result.docId, result.pageNum || 1);
           }
         }
       }

       return results;
     }
     ```

  4. Update index.js exports to include HybridSearchService

  Do NOT remove existing visualSearch method - keep for backwards compatibility.

<task type="auto">
  <name>Task 2: Create hybrid search API endpoint</name>
  <files>routes/api/visual-rag.js (new), routes/index.js</files>
  <action>
  Create a new API route for hybrid search:

  routes/api/visual-rag.js:
  ```javascript
  const express = require('express');
  const router = express.Router();
  const { ingestionManager } = require('../../services/visual-rag');

  // POST /api/visual-rag/search
  router.post('/search', async (req, res) => {
    try {
      const { query, k = 5, mode = 'hybrid', includeOverlays = true } = req.body;

      if (!query) {
        return res.status(400).json({ error: 'Query is required' });
      }

      const results = await ingestionManager.hybridSearch(query, {
        k,
        mode,
        includeOverlays
      });

      res.json({
        success: true,
        query,
        mode,
        ...results
      });
    } catch (error) {
      res.status(500).json({
        success: false,
        error: error.message
      });
    }
  });

  // GET /api/visual-rag/health
  router.get('/health', async (req, res) => {
    const health = await ingestionManager.healthCheck();
    res.json(health);
  });

  module.exports = router;
  ```

  Update routes/index.js to mount at /api/visual-rag

  Include mode parameter: 'hybrid' (default), 'visual', 'text' for comparison testing.
