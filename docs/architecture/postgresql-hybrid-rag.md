<system_profile>
    <id>postgresql-hybrid-rag</id>
    <name>PostgreSQL Hybrid Visual RAG</name>
    <status>Active Strategy</status>

    <core_components>
        <database>PostgreSQL 16+ (Existing Paperless DB)</database>
        <extension>pgvector (Vector Similarity Search)</extension>
        <visual_model>Qwen3-VL 8B (via Ollama)</visual_model>
        <embedding_model>nomic-embed-text-v1.5 (via Ollama)</embedding_model>
    </core_components>

    <storage_schema>
        <table name="document_embeddings">
            <column name="doc_id" type="ForeignKey">Link to paperless_documents</column>
            <column name="embedding" type="vector(768)">Semantic representation of Qwen's visual description</column>
            <column name="visual_summary" type="TEXT">The text description generated by Qwen3-VL</column>
            <column name="bounding_boxes" type="JSONB">
                Structure: [{"label": "signature", "box": [x1, y1, x2, y2], "confidence": 0.95}]
            </column>
            <column name="comparison_diff" type="JSONB">
                Differences between Tesseract OCR and Qwen Visual findings.
            </column>
        </table>
    </storage_schema>

    <workflow_logic>
        <ingestion_trigger>
            Post-consumption script triggers Qwen3-VL analysis.
        </ingestion_trigger>
        <processing_step>
            1. Qwen3-VL scans image at 300 DPI.
            2. Qwen generates description + coordinates (e.g., "Signature at [100,200...]").
            3. 'nomic-embed-text' converts description to Vector.
            4. Data is INSERTED into 'document_embeddings' in a single transaction.
        </processing_step>
        <retrieval_step>
            SQL Query joins 'paperless_documents' (Metadata) with 'document_embeddings' (Vectors).
            Returns: Text Answer + JSON Coordinates for frontend overlay.
        </retrieval_step>
    </workflow_logic>

    <hardware_validation>
        <gpu>RTX 3090 Ti (24GB)</gpu>
        <allocation>
            - Qwen3-VL: ~6.5 GB
            - Chat Model: ~5.5 GB
            - Overhead/Context: ~12 GB (Plenty for high-res analysis)
        </allocation>
    </hardware_validation>
</system_profile>
